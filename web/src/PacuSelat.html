/* * PacuSelat.html — Modal Pacu Selat dengan scroll lock untuk iOS * Canvas, HUD, efek, menara peringkat, kartu hasil */ (function() { window.MTGames8 = class PacuSelat { constructor(config = {}) { this.config = config; this.canvas = null; this.ctx = null; this.running = false; this.finished = false; this.init(); } init() { // Buat atau ambil canvas this.canvas = document.getElementById('pacu-canvas') || this.createCanvas(); this.ctx = this.canvas.getContext('2d'); // === SCROLL LOCK: Tambah touchmove listener ke canvas === this.canvas.addEventListener('touchmove', (e) => { // Cegah scroll saat touch canvas di iOS e.preventDefault(); }, { passive: false }); // Additional touch handlers untuk mencegah scroll this.canvas.addEventListener('touchstart', (e) => { this.lastTouchY = e.touches[0].clientY; }, { passive: false }); this.canvas.addEventListener('touchend', (e) => { delete this.lastTouchY; }, { passive: false }); // Setup responsive canvas this.resize(); window.addEventListener('resize', () => this.resize()); // Setup game state this.reset(); } createCanvas() { const canvas = document.createElement('canvas'); canvas.id = 'pacu-canvas'; canvas.style.cssText = ` display: block; width: 100%; height: 100%; background: linear-gradient(to bottom, #87ceeb, #0a4c7a); touch-action: none; `; const modal = document.getElementById('pacu-modal') || document.body; modal.appendChild(canvas); return canvas; } resize() { if (!this.canvas) return; const dpr = window.devicePixelRatio || 1; const rect = this.canvas.parentElement.getBoundingClientRect(); this.canvas.width = rect.width * dpr; this.canvas.height = rect.height * dpr; this.ctx.scale(dpr, dpr); this.width = rect.width; this.height = rect.height; } reset() { this.position = 0; this.speed = 0; this.gear = 1; this.heat = 0; this.startTime = Date.now(); this.paused = false; this.finished = false; } start() { this.running = true; this.reset(); this.loop(); // Delegate ke host if (window.PacuSelatHost?.onStart) { window.PacuSelatHost.onStart(this); } } finish() { this.finished = true; this.running = false; const result = { time: Date.now() - this.startTime, position: 1, competitors: 4, personal_best: false }; if (window.PacuSelatHost?.onFinish) { window.PacuSelatHost.onFinish(result); } } close() { this.running = false; this.finished = false; if (window.PacuSelatHost?.onClose) { window.PacuSelatHost.onClose(); } if (this.canvas) { this.canvas.remove(); this.canvas = null; } } loop = () => { if (!this.running) return; this.update(); this.draw(); requestAnimationFrame(this.loop); } update() { // Simulasi fisika kapal const now = Date.now() - this.startTime; this.position = Math.min(now / 50, 100); // Jarak finish 100% if (this.position >= 100) { this.finish(); } } draw() { // Clear canvas this.ctx.fillStyle = 'rgba(135, 206, 235, 1)'; this.ctx.fillRect(0, 0, this.width, this.height); // Draw track this.drawTrack(); // Draw ship this.drawShip(); // Draw HUD this.drawHUD(); } drawTrack() { const y = this.height * 0.6; const trackWidth = this.width * 0.9; const trackX = this.width * 0.05; this.ctx.strokeStyle = '#fff'; this.ctx.lineWidth = 3; this.ctx.beginPath(); this.ctx.moveTo(trackX, y); this.ctx.lineTo(trackX + trackWidth, y); this.ctx.stroke(); // Progress bar const progressX = trackX + (trackWidth * this.position / 100); this.ctx.fillStyle = '#228b22'; this.ctx.fillRect(trackX, y - 10, progressX - trackX, 20); } drawShip() { const y = this.height * 0.6; const trackWidth = this.width * 0.9; const trackX = this.width * 0.05; const x = trackX + (trackWidth * this.position / 100); // Draw simple ship shape this.ctx.fillStyle = '#d4af37'; this.ctx.beginPath(); this.ctx.arc(x, y, 15, 0, Math.PI * 2); this.ctx.fill(); // Draw sail this.ctx.fillStyle = '#fff'; this.ctx.beginPath(); this.ctx.moveTo(x, y - 15); this.ctx.lineTo(x + 20, y - 25); this.ctx.lineTo(x + 20, y - 5); this.ctx.closePath(); this.ctx.fill(); } drawHUD() { this.ctx.fillStyle = '#2c1810'; this.ctx.font = 'bold 16px Lato'; this.ctx.fillText(`Waktu: ${((Date.now() - this.startTime) / 1000).toFixed(2)}s`, 20, 30); this.ctx.fillText(`Jarak: ${this.position.toFixed(0)}%`, 20, 60); } onInput(key, value) { switch(key) { case 'gear': this.gear = value; break; case 'throttle': this.speed = value; break; } } onJempol(side) { console.log('Jempol:', side); } }; // Start game saat diminta window.startPacuSelat = () => { const game = new MTGames8(); game.start(); return game; }; })();
